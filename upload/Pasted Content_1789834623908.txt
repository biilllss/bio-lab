function taperedTube(pts, radiusFn, material, tubular=64, radial=16){
  const curve = new THREE.CatmullRomCurve3(pts.map(p=>V3(...p)));
  const frames = curve.computeFrenetFrames(tubular, false);
  const pos=[], nor=[], uvs=[], idx=[];
  for(let i=0;i<=tubular;i++){
    const u=i/tubular, P=curve.getPointAt(u);
    const N=frames.normals[i], B=frames.binormals[i], r=radiusFn(u);
    for(let j=0;j<=radial;j++){
      const v=j/radial*Math.PI*2, sin=Math.sin(v), cos=-Math.cos(v);
      const nx=cos*N.x+sin*B.x, ny=cos*N.y+sin*B.y, nz=cos*N.z+sin*B.z;
      pos.push(P.x+r*nx, P.y+r*ny, P.z+r*nz);
      nor.push(nx,ny,nz);
      uvs.push(j/radial, i/tubular);
    }
  }
  for(let i=1;i<=tubular;i++) for(let j=1;j<=radial;j++){
    const a=(radial+1)*(i-1)+(j-1), b=(radial+1)*i+(j-1),
          c=(radial+1)*i+j,       d=(radial+1)*(i-1)+j;
    idx.push(a,b,d, b,c,d);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('normal',   new THREE.Float32BufferAttribute(nor,3));
  g.setAttribute('uv',       new THREE.Float32BufferAttribute(uvs,2));
  g.setIndex(idx);
  return new THREE.Mesh(g, material);
}
function organic(geo, amp, freq){
  if(amp<=0){ geo.computeVertexNormals(); return geo; }
  const p=geo.attributes.position;
  for(let i=0;i<p.count;i++){
    const x=p.getX(i), y=p.getY(i), z=p.getZ(i);
    const d=1+amp*(Math.sin(x*freq+1.3)+Math.sin(y*freq*1.7+2.1)+Math.sin(z*freq*1.3+4.2));
    p.setXYZ(i, x*d, y*d, z*d);
  }
  geo.computeVertexNormals();
  return geo;
}

function ribbon(ptsA, ptsB, material, segs=20){
  const cA=new THREE.CatmullRomCurve3(ptsA.map(p=>V3(...p)));
  const cB=new THREE.CatmullRomCurve3(ptsB.map(p=>V3(...p)));
  const pos=[], uv=[], idx=[];
  for(let i=0;i<=segs;i++){
    const u=i/segs, pa=cA.getPointAt(u), pb=cB.getPointAt(u);
    pos.push(pa.x,pa.y,pa.z, pb.x,pb.y,pb.z);
    uv.push(u,0, u,1);
  }
  for(let i=0;i<segs;i++){
    const a=i*2,b=i*2+1,c=(i+1)*2,d=(i+1)*2+1;
    idx.push(a,c,b, c,d,b);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',       new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(idx); g.computeVertexNormals();
  return new THREE.Mesh(g, material);
}

function makeSys(prefix){
  const sys = { prefix, group:new THREE.Group(), parts:{}, pick:[],
    order:[], quizKeys:[], inner:new Set(), labelObjs:[], labelLines:[],
    labelMeta:[], mats:[], camPos:V3(0,2,8), camTarget:V3(0,0,0) };
  scene.add(sys.group);
  sys.register = (key, mesh, parent)=>{
    mesh.userData.part = key;
    mesh.castShadow = true; mesh.receiveShadow = true;
    (sys.parts[key] ||= []).push(mesh);
    (partMeshes[key] ||= []).push(mesh);
    sys.pick.push(mesh);
    (parent || sys.group).add(mesh);
    return mesh;
  };
  sys.add = (key, mesh)=> sys.register(prefix+key, mesh);
  sys.addDecor = m => { sys.group.add(m); return m; };
  return sys;
}

const _p=new THREE.Vector3(), _t=new THREE.Vector3(), _s=new THREE.Vector3(),
      _fwd=new THREE.Vector3(0,0,1), _up=new THREE.Vector3(0,1,0);
function updateSperm(t){
  for(const set of spermSets) for(const sp of set.sperms){
    const u=(t*0.06+sp.phase)%1;
    set.curve.getPointAt(u,_p); set.curve.getTangentAt(u,_t);
    sp.head.position.copy(_p); sp.glow.position.copy(_p);
    sp.head.quaternion.setFromUnitVectors(_fwd,_t);
    const pos=sp.tail.geometry.attributes.position;
    for(let k=0;k<TAIL_N;k++){
      const uu=Math.max(0, u-k*set.tailStep);
      set.curve.getPointAt(uu,_p); set.curve.getTangentAt(uu,_t);
      _s.crossVectors(_t,_up);
      if(_s.lengthSq()<1e-6) _s.set(1,0,0); else _s.normalize();
      const w=Math.sin(t*9-k*0.9+sp.phase*12.57)*0.024*(k/(TAIL_N-1));
      pos.setXYZ(k,_p.x+_s.x*w,_p.y+_s.y*w,_p.z+_s.z*w);
    }
    pos.needsUpdate=true;
  }
}

  for(const k in partMeshes){

      continue;
    }
    const lvl=(!quiz.active && k===selectedKey)?0.28:(!quiz.active && k===hoverKey)?0.14:0;
    m.emissive.setScalar(lvl);
  }
  for(const p of pulses){
    const u=(t-p.t0)/p.dur;
    if(u<0||u>1) continue;
    const w=Math.sin(Math.PI*u)*0.75;
    const m=partMeshes[p.key][0].material;
    m.emissive.r=Math.max(m.emissive.r,p.c.r*w);
    m.emissive.g=Math.max(m.emissive.g,p.c.g*w);
    m.emissive.b=Math.max(m.emissive.b,p.c.b*w);
  }
  for(let i=pulses.length-1;i>=0;i--)
    if(t-pulses[i].t0>pulses[i].dur) pulses.splice(i,1);
}

    const cls = i<quiz.idx ? (quiz.results[i]==='ok'?'ok':'bad')
              : (i===quiz.idx?'cur':'');
    return `<i class="${cls}" title="${INFO[k].name}"></i>`;
  }).join('')+'</div>';
}
function showQuizStart(){
  panel.classList.toggle('fem', active==='female');
  panel.innerHTML=`
    <div class="p-name">🎯 Quiz</div>
    <div class="p-sub">${active==='male'?'♂ MALE':'♀ FEMALE'} · pick a mode</div>
    <div class="q-opts">
      <button data-act="qname"><b>Name mode</b><br>A part glows — pick its name</button>
      <button data-act="qfind"><b>Find mode</b><br>I name a part — click it on the model</button>
      <button data-act="qmix"><b>Mixed mode</b><br>Both systems · 24 questions</button>
    </div>
    <div class="q-actions"><button data-act="qcancel">Cancel</button></div>`;
}
function startQuiz(mode, keys){
  quiz.active=true; quiz.mode=mode; quiz.locked=false;
  quiz.system=active;
  if(!keys) keys = mode==='mixed'
    ? SYSTEMS.male.quizKeys.concat(SYSTEMS.female.quizKeys)
    : SYSTEMS[active].quizKeys;
  quiz.order=shuffle([...keys]); quiz.idx=0; quiz.results=[];
  quiz.t0=performance.now();
  quiz.prevLabels=labelsOn;
  setLabels(false);
  xrayQuiz=(mode==='name'); applyXray();
  controls.autoRotate=false;
  selectedKey=null; hoverKey=null;
  renderQuestion();
}
function renderQuestion(){
  if(quiz.idx>=quiz.order.length){ endQuiz(); return; }
  const key=quiz.order[quiz.idx];
  if(quiz.mode==='mixed'){ if(sysOf(key)!==active) showSystem(sysOf(key),{silent:true}); }
  else if(active!==quiz.system){ showSystem(quiz.system,{silent:true}); }
  if(quiz.mode==='find'||quiz.mode==='mixed'){
    const inner=(SYSTEMS[quiz.system].inner.has(key))
      ? '<div class="q-hint">💡 It sits inside/behind another part — click straight through the outer layer.</div>':'';
    const label = quiz.mode==='mixed' ? 'Quiz · Mixed' : 'Quiz · Find';
    panel.innerHTML=`
      <div class="q-head"><span>${label}</span><span id="qtime">0:00</span></div>
      <div class="q-big">Click the: <b>${INFO[key].name}</b></div>
      ${inner}${dotsHtml()}
      <div class="q-fb" id="qfb">&nbsp;</div>
      <div class="q-actions">
        <button data-act="qskip">Skip</button>
        <button data-act="qend">End quiz</button>
      </div>`;
    marker.visible=false;
  } else {
    const opts=shuffle([key, ...sample(
      (quiz.mode==='mixed'?SYSTEMS.male.quizKeys.concat(SYSTEMS.female.quizKeys)
        :SYSTEMS[active].quizKeys).filter(k=>k!==key),3)]);
    panel.innerHTML=`
      <div class="q-head"><span>Quiz · Name</span><span id="qtime">0:00</span></div>
      <div class="q-big">Which structure is <b>glowing ❓</b></div>
      ${dotsHtml()}
      <div class="q-opts">${opts.map(k=>
        `<button data-act="qopt" data-key="${k}">${INFO[k].name}</button>`).join('')}</div>
      <div class="q-fb" id="qfb">&nbsp;</div>
      <div class="q-actions">
        <button data-act="qskip">Skip</button>
        <button data-act="qend">End quiz</button>
      </div>`;
    marker.visible=true;
    marker.position.copy(PART_CENTER[key]).add(partOffset(key, expCur));
    focusPart(key);
  }
}
function nextQ(){ quiz.locked=false; quiz.idx++; renderQuestion(); }
function resolveAnswer(correct, target, wrongKey){
  if(correct){
    quiz.results[quiz.idx]='ok';
    fb('✔ Correct!','ok');
    pulse(target,0x2ecc71,1.1); sndOk();
    quiz.locked=true; setTimeout(nextQ, 900);
  } else {
    quiz.results[quiz.idx]='bad';
    const wrong=wrongKey && INFO[wrongKey] ? INFO[wrongKey].name : 'empty space';
    fb(`✘ That was <b>${wrong}</b> — the <b>${INFO[target].name}</b> is here →`,'bad');
    if(wrongKey) pulse(wrongKey,0xe74c3c,1.1);
    pulse(target,0x2ecc71,1.9);
    focusPart(target); sndBad();
    quiz.locked=true; setTimeout(nextQ, 2200);
  }
}
function quizFind(hits){
  if(quiz.locked) return;
  const target=quiz.order[quiz.idx];
  const first=hits.find(k=>!QUIZ_SKIP.has(k));
  const inner=SYSTEMS[quiz.system].inner.has(target);
  const correct=inner ? hits.includes(target) : first===target;
  resolveAnswer(correct, target, first);
}
function quizAnswerName(chosen){
  if(quiz.locked) return;
  const target=quiz.order[quiz.idx];
  resolveAnswer(chosen===target, target, chosen);
}
function quizSkip(){
  if(quiz.locked) return;
  const target=quiz.order[quiz.idx];
  quiz.results[quiz.idx]='bad';
  fb(`→ The <b>${INFO[target].name}</b> is here`,'');
  pulse(target,0x2ecc71,1.5); focusPart(target);
  quiz.locked=true; setTimeout(nextQ, 1500);
}
function endQuiz(){
  quiz.active=false; quiz.locked=false;
  marker.visible=false;
  xrayQuiz=false; applyXray();
  setLabels(quiz.prevLabels);
  const n=quiz.order.length;
  const ok=quiz.results.filter(r=>r==='ok').length;
  const missed=quiz.order.filter((k,i)=>quiz.results[i]!=='ok');
  quiz.lastMissed=missed; quiz.lastMode=quiz.mode;
  const secs=Math.round((performance.now()-quiz.t0)/1000);
  panel.innerHTML=`
    <div class="p-name">📋 Results</div>
    <div class="p-sub">${ok} / ${n} correct · ${Math.round(ok/n*100)}% · ${fmtTime(secs)}</div>
    <div class="q-dots">${quiz.order.map((k,i)=>{
      const cls=quiz.results[i]==='ok'?'ok':quiz.results[i]?'bad':'';
      return `<i class="${cls}" title="${INFO[k].name}"></i>`;}).join('')}</div>
    ${missed.length
      ? `<div class="p-fn" style="margin-top:8px"><b>Review:</b> ${
          missed.map(k=>`<button class="chip" data-act="chip" data-key="${k}">${INFO[k].name}</button>`).join(' ')}</div>`
      : '<div class="q-fb ok">Perfect! 🎉</div>'}
    <div class="q-actions">
      ${missed.length?'<button data-act="qretry">Retry missed</button>':''}
      <button data-act="qnew">New quiz</button>
      <button data-act="qcancel">Done</button>
    </div>`;
  sndEnd();
}

  labelsOn=on;
  applyLabelsVis();

  btnLabels.classList.toggle('on', !on);
}

  if(!silent && quiz.active) endQuiz();
  active=name;
  SYSTEMS.male.group.visible = name==='male';
  SYSTEMS.female.group.visible = name==='female';
  applyLabelsVis();
  spermGroup.visible=false; eggGroup.visible=false; xrayPath=false;
  applyXray(); updatePathBtn();
  setExplode(0);
  document.getElementById('sysSub').textContent = name==='male'
    ? '♂ Male — drag to rotate · scroll to zoom · click parts'
    : '♀ Female — drag to rotate · scroll to zoom · click parts';
  hideMenu();

    camera.position.copy(sys.camPos);
    controls.target.copy(sys.camTarget);

    showDefault();
  }
}

  if(e.repeat) return;
  const k=e.key.toLowerCase();
  if(k==='1') showSystem('male');
  else if(k==='2') showSystem('female');
  else if(k==='b'||e.key==='Escape'){
    menu.classList.contains('hidden') ? showMenu() : showSystem(active);
  }
  else if(k==='l') btnLabels.click();
  else if(k==='x') btnXray.click();
  else if(k==='s') btnSperm.click();
  else if(k==='c') btnCut.click();

  else if(k==='q') btnQuiz.click();
  else if(k==='r') document.getElementById('btnReset').click();
  else if(k==='m') btnSound.click();
});

const clock=new THREE.Clock();
(function animate(){
  requestAnimationFrame(animate);
  const t=clock.getElapsedTime(); simT=t;
  controls.update();
  if(focusGoal && performance.now()<focusUntil)
    controls.target.lerp(focusGoal, 0.08);
  if(Math.abs(expCur-expTarget)>0.001){
    expCur+=(expTarget-expCur)*0.12;
    if(Math.abs(expCur-expTarget)<0.001) expCur=expTarget;
    applyExplode(expCur);
  }
  updateEmissives(t);
  if(spermGroup.visible){ updateSperm(t); guideMatSperm.opacity=0.22+0.1*Math.sin(t*3); }
  if(eggGroup.visible){ updateEgg(t); guideMatEgg.opacity=0.25+0.12*Math.sin(t*3); }
  if(quiz.active){
    const el=document.getElementById('qtime');
    if(el) el.textContent=fmtTime((performance.now()-quiz.t0)/1000|0);
  }
  renderer.render(scene, camera);
  labelRenderer.render(scene, camera);
})();

  renderer.setSize(innerWidth, innerHeight);
  labelRenderer.setSize(innerWidth, innerHeight);
});
